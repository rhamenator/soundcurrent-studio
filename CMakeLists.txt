cmake_minimum_required(VERSION 3.20)
project(soundcurrent-studio VERSION 1.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SOUNDCURRENT_BUILD_DESKTOP "Build the Qt desktop equalizer" ON)
if(SOUNDCURRENT_BUILD_DESKTOP)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Network Widgets)
add_library(soundcurrent-studio-ui STATIC src/studio_model.cpp src/studio_panel.cpp)
target_link_libraries(soundcurrent-studio-ui PUBLIC soundcurrent-wave Qt6::Widgets)
if(WIN32)
  enable_language(RC)
  add_executable(soundcurrent-studio WIN32 src/main.cpp src/equipment_profiles.cpp src/processing_guard.cpp data/resources.qrc data/localization/resources.qrc data/windows.rc)
  target_include_directories(soundcurrent-studio PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/data)
  target_link_libraries(soundcurrent-studio PRIVATE soundcurrent-windows-audio soundcurrent-studio-ui Qt6::Widgets Qt6::Network)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3)
  add_library(soundcurrent-linux-audio STATIC src/linux_audio.cpp)
  target_link_libraries(soundcurrent-linux-audio PUBLIC soundcurrent-engine PkgConfig::PIPEWIRE)
  add_executable(soundcurrent-linux-audio-smoke tests/linux_audio_smoke.cpp)
  target_link_libraries(soundcurrent-linux-audio-smoke PRIVATE soundcurrent-linux-audio Qt6::Core)
  add_executable(soundcurrent-studio src/main.cpp src/equipment_profiles.cpp src/processing_guard.cpp data/resources.qrc data/localization/resources.qrc)
  target_link_libraries(soundcurrent-studio PRIVATE soundcurrent-linux-audio soundcurrent-studio-ui Qt6::Widgets Qt6::Network)
endif()
if(MSVC)
  target_compile_options(soundcurrent-studio PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(soundcurrent-studio PRIVATE -Wall -Wextra -Wpedantic)
endif()

endif() # SOUNDCURRENT_BUILD_DESKTOP

add_library(soundcurrent-dsp STATIC src/dsp.cpp src/enhancement.cpp)
target_include_directories(soundcurrent-dsp PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
  $<INSTALL_INTERFACE:include/soundcurrent>)
target_compile_features(soundcurrent-dsp PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(soundcurrent-dsp PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(soundcurrent-dsp PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(soundcurrent-engine STATIC src/engine.cpp)
add_library(SoundCurrent::Engine ALIAS soundcurrent-engine)
target_link_libraries(soundcurrent-engine PUBLIC soundcurrent-dsp)
target_include_directories(soundcurrent-engine PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
  $<INSTALL_INTERFACE:include/soundcurrent>)
set_target_properties(soundcurrent-engine PROPERTIES EXPORT_NAME Engine POSITION_INDEPENDENT_CODE ON)
set_target_properties(soundcurrent-dsp PROPERTIES EXPORT_NAME Dsp POSITION_INDEPENDENT_CODE ON)
add_library(soundcurrent-wave STATIC src/wav.cpp)
add_library(SoundCurrent::Wave ALIAS soundcurrent-wave)
set_target_properties(soundcurrent-wave PROPERTIES EXPORT_NAME Wave POSITION_INDEPENDENT_CODE ON)
target_link_libraries(soundcurrent-wave PUBLIC soundcurrent-engine)
add_executable(soundcurrent-studio-render src/studio_render.cpp)
target_link_libraries(soundcurrent-studio-render PRIVATE soundcurrent-wave)
add_executable(soundcurrent-engine-test tests/engine_response.cpp)
target_link_libraries(soundcurrent-engine-test PRIVATE soundcurrent-engine)
foreach(target soundcurrent-engine soundcurrent-wave soundcurrent-studio-render soundcurrent-engine-test)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
add_executable(soundcurrent-dsp-test tests/dsp_response.cpp)
target_link_libraries(soundcurrent-dsp-test PRIVATE soundcurrent-dsp)
if(WIN32 AND MINGW)
  target_link_options(soundcurrent-dsp-test PRIVATE -static-libstdc++ -static-libgcc)
endif()
enable_testing()

if(WIN32 AND SOUNDCURRENT_BUILD_DESKTOP)
  add_subdirectory(native/windows/driver-manager)
  # Opt-in cross-user Windows mutex test, with no Qt/User32 startup dependency.
  add_executable(soundcurrent-windows-session-gate-smoke tests/windows_session_gate_smoke.cpp)
  target_include_directories(soundcurrent-windows-session-gate-smoke PRIVATE src)
  target_link_libraries(soundcurrent-windows-session-gate-smoke PRIVATE advapi32)
  set_target_properties(soundcurrent-windows-session-gate-smoke PROPERTIES AUTOMOC OFF)
  if(MSVC)
    set_target_properties(soundcurrent-windows-session-gate-smoke PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreaded)
  endif()

  add_library(soundcurrent-windows-audio STATIC src/windows_audio.cpp src/windows_io.cpp)
  target_include_directories(soundcurrent-windows-audio PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
  target_link_libraries(soundcurrent-windows-audio PUBLIC soundcurrent-engine ole32 uuid avrt advapi32)
  if(MINGW)
    target_link_libraries(soundcurrent-windows-audio PUBLIC ksguid)
  endif()
  add_executable(soundcurrent-route-guardian src/windows_route_guardian.cpp)
  target_link_libraries(soundcurrent-route-guardian PRIVATE soundcurrent-windows-audio shell32)
  if(MINGW)
    target_link_options(soundcurrent-route-guardian PRIVATE -static)
  endif()
  install(TARGETS soundcurrent-route-guardian RUNTIME DESTINATION bin)
  add_executable(soundcurrent-cable-setup-guard src/windows_cable_setup_guard.cpp)
  target_link_libraries(soundcurrent-cable-setup-guard PRIVATE soundcurrent-windows-audio Qt6::Core shell32)
  install(TARGETS soundcurrent-cable-setup-guard RUNTIME DESTINATION bin)
  add_executable(soundcurrent-windows-route-contract tests/windows_route_contract.cpp)
  target_link_libraries(soundcurrent-windows-route-contract PRIVATE soundcurrent-windows-audio)
  if(MINGW)
    target_link_options(soundcurrent-windows-route-contract PRIVATE -static-libstdc++ -static-libgcc)
  endif()
  add_test(NAME windows-route-contract COMMAND soundcurrent-windows-route-contract)
  # Opt-in test changes only the clone's audio routes. It kills its own child
  # to check the same managed lease and guardian used by the desktop app.
  add_executable(soundcurrent-windows-route-smoke tests/windows_route_smoke.cpp)
  target_link_libraries(soundcurrent-windows-route-smoke PRIVATE soundcurrent-windows-audio Qt6::Core)
  add_dependencies(soundcurrent-windows-route-smoke soundcurrent-route-guardian)
  # Opt-in guest endpoint volume check; does not generate a test signal.
  add_executable(soundcurrent-windows-volume-smoke tests/windows_volume_route_smoke.cpp)
  target_link_libraries(soundcurrent-windows-volume-smoke PRIVATE soundcurrent-windows-audio Qt6::Core)
  add_dependencies(soundcurrent-windows-volume-smoke soundcurrent-route-guardian)

  # This integration check plays audio only with --run; keep it out of CTest.
  add_executable(soundcurrent-windows-audio-smoke tests/windows_audio_smoke.cpp)
  target_link_libraries(soundcurrent-windows-audio-smoke PRIVATE soundcurrent-windows-audio)
  if(MINGW)
    target_link_options(soundcurrent-windows-audio-smoke PRIVATE -static-libstdc++ -static-libgcc)
  endif()
endif()
