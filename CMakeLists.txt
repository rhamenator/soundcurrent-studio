cmake_minimum_required(VERSION 3.20)
project(soundcurrent-eq VERSION 0.7.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SOUNDCURRENT_BUILD_DESKTOP "Build the Qt desktop equalizer" ON)
if(SOUNDCURRENT_BUILD_DESKTOP)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Network Widgets)
if(WIN32)
  enable_language(RC)
  add_executable(soundcurrent-eq WIN32 src/main.cpp data/resources.qrc data/windows.rc)
  target_include_directories(soundcurrent-eq PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/data)
  target_link_libraries(soundcurrent-eq PRIVATE soundcurrent-windows-audio Qt6::Widgets Qt6::Network)
else()
  add_executable(soundcurrent-eq src/main.cpp data/resources.qrc)
  target_link_libraries(soundcurrent-eq PRIVATE soundcurrent-dsp Qt6::Widgets Qt6::Network)
endif()
if(MSVC)
  target_compile_options(soundcurrent-eq PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(soundcurrent-eq PRIVATE -Wall -Wextra -Wpedantic)
endif()

endif() # SOUNDCURRENT_BUILD_DESKTOP

add_library(soundcurrent-dsp STATIC src/dsp.cpp)
target_include_directories(soundcurrent-dsp PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
  $<INSTALL_INTERFACE:include/soundcurrent>)
target_compile_features(soundcurrent-dsp PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(soundcurrent-dsp PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(soundcurrent-dsp PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(soundcurrent-engine STATIC src/engine.cpp)
add_library(SoundCurrent::Engine ALIAS soundcurrent-engine)
target_link_libraries(soundcurrent-engine PUBLIC soundcurrent-dsp)
target_include_directories(soundcurrent-engine PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
  $<INSTALL_INTERFACE:include/soundcurrent>)
set_target_properties(soundcurrent-engine PROPERTIES EXPORT_NAME Engine POSITION_INDEPENDENT_CODE ON)
set_target_properties(soundcurrent-dsp PROPERTIES EXPORT_NAME Dsp POSITION_INDEPENDENT_CODE ON)
add_library(soundcurrent-wave STATIC src/wav.cpp)
add_library(SoundCurrent::Wave ALIAS soundcurrent-wave)
set_target_properties(soundcurrent-wave PROPERTIES EXPORT_NAME Wave POSITION_INDEPENDENT_CODE ON)
target_link_libraries(soundcurrent-wave PUBLIC soundcurrent-engine)
add_executable(soundcurrent-studio-render src/studio_render.cpp)
target_link_libraries(soundcurrent-studio-render PRIVATE soundcurrent-wave)
add_executable(soundcurrent-engine-test tests/engine_response.cpp)
target_link_libraries(soundcurrent-engine-test PRIVATE soundcurrent-engine)
foreach(target soundcurrent-engine soundcurrent-wave soundcurrent-studio-render soundcurrent-engine-test)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
add_executable(soundcurrent-dsp-test tests/dsp_response.cpp)
target_link_libraries(soundcurrent-dsp-test PRIVATE soundcurrent-dsp)
if(WIN32 AND MINGW)
  target_link_options(soundcurrent-dsp-test PRIVATE -static-libstdc++ -static-libgcc)
endif()
enable_testing()
add_test(NAME dsp-response COMMAND soundcurrent-dsp-test)
add_test(NAME studio-engine COMMAND soundcurrent-engine-test)
set_tests_properties(studio-engine PROPERTIES TIMEOUT 60)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME studio-wave COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/studio_wave.py $<TARGET_FILE:soundcurrent-studio-render>)
  set_tests_properties(studio-wave PROPERTIES TIMEOUT 60)
endif()
if(SOUNDCURRENT_BUILD_DESKTOP)
add_test(NAME shared-ui COMMAND soundcurrent-eq --ui-self-test)
set_tests_properties(shared-ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 45)
endif()

if(WIN32 AND SOUNDCURRENT_BUILD_DESKTOP)
  add_library(soundcurrent-windows-audio STATIC src/windows_audio.cpp src/windows_io.cpp)
  target_include_directories(soundcurrent-windows-audio PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
  target_link_libraries(soundcurrent-windows-audio PUBLIC soundcurrent-dsp ole32 uuid avrt)
  if(MINGW)
    target_link_libraries(soundcurrent-windows-audio PUBLIC ksguid)
  endif()
  # This integration check plays audio only with --run; keep it out of CTest.
  add_executable(soundcurrent-windows-audio-smoke tests/windows_audio_smoke.cpp)
  target_link_libraries(soundcurrent-windows-audio-smoke PRIVATE soundcurrent-windows-audio)
  if(MINGW)
    target_link_options(soundcurrent-windows-audio-smoke PRIVATE -static-libstdc++ -static-libgcc)
  endif()
endif()

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
install(TARGETS soundcurrent-engine soundcurrent-dsp soundcurrent-wave EXPORT SoundCurrentEngineTargets
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES src/engine.h src/dsp.h src/wav.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/soundcurrent)
install(EXPORT SoundCurrentEngineTargets NAMESPACE SoundCurrent::
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SoundCurrentEngine)
configure_package_config_file(cmake/SoundCurrentEngineConfig.cmake.in
  ${CMAKE_CURRENT_BINARY_DIR}/SoundCurrentEngineConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SoundCurrentEngine)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/SoundCurrentEngineConfigVersion.cmake
  VERSION 0.1.0 COMPATIBILITY SameMinorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/SoundCurrentEngineConfig.cmake
  ${CMAKE_CURRENT_BINARY_DIR}/SoundCurrentEngineConfigVersion.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SoundCurrentEngine)
install(TARGETS soundcurrent-studio-render RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(SOUNDCURRENT_BUILD_DESKTOP)
install(TARGETS soundcurrent-eq RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(NOT WIN32)
  install(FILES data/io.github.rhamenator.SoundCurrentEQ.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  install(FILES data/io.github.rhamenator.SoundCurrentEQ.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
endif()
endif()
install(FILES COPYRIGHT DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/soundcurrent-eq RENAME copyright)
install(FILES THIRD-PARTY-NOTICES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/soundcurrent-eq)
install(FILES docs/studio-engine.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/soundcurrent-eq)
install(DIRECTORY data/speakers DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/soundcurrent-eq)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/soundcurrent-eq)
